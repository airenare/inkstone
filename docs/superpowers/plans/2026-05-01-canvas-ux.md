# Canvas UX Upgrade Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Upgrade InkStone's canvas from a basic static layout to an interactive, jsoncanvas-spec-compliant experience with pan/zoom, full markdown in text nodes, correct edge directionality, and visual polish.

**Architecture:** A new `canvas-stage` div wraps all nodes and the SVG edge layer inside the existing `canvas-view` viewport. JavaScript applies `transform: translate() scale()` to the stage for pan/zoom. Node positions switch from percentage-based to absolute pixel values matching canvas coordinates. Text nodes run through the full `render_markdown()` pipeline. Edges respect `fromEnd`/`toEnd` per the jsoncanvas spec.

**Tech Stack:** Python (canvas.py, converters.py), vanilla JS (~100 lines inline in base.html), CSS (base.css). No external libraries.

```python
def _canvas_text_to_html(text):
    """Minimal markdown rendering for canvas text nodes."""
    escaped = _html.escape(text)
    parts = []
    for line in escaped.split("\n"):
        line = re.sub(r"\*\*(.+?)\*\*", r"<strong>\1</strong>", line)
        line = re.sub(r"__(.+?)__", r"<strong>\1</strong>", line)
        line = re.sub(r"\*(.+?)\*", r"<em>\1</em>", line)
        line = re.sub(r"`(.+?)`", r"<code>\1</code>", line)
        parts.append(line)
    return "<br>".join(parts)
```

## Task 7: `base.html` — `initCanvas()` JavaScript

**Files:**
- Modify: `frontend/templates/base.html` (add `<script>` block before `</body>`)

- [ ] **Step 1: Append the canvas JS block to `base.html`**

Find the last `</body>` tag in `frontend/templates/base.html` and insert a new `<script>` block immediately before it:

```html
<script>
(function() {
    function initCanvas(view) {
        var stage = view.querySelector(".canvas-stage");
        var fitBtn = view.querySelector(".canvas-fit-btn");
        if (!stage) return;

        var tx = 0, ty = 0, scale = 1;
        var dragging = false, lastX = 0, lastY = 0;

        function clampScale(s) {
            return Math.max(0.2, Math.min(4, s));
        }

        function applyTransform() {
            stage.style.transform =
                "translate(" + tx + "px," + ty + "px) scale(" + scale + ")";
        }

        function fitToView() {
            var vw = view.clientWidth, vh = view.clientHeight;
            var sw = parseFloat(stage.style.width);
            var sh = parseFloat(stage.style.height);
            scale = clampScale(Math.min(vw / sw, vh / sh) * 0.9);
            tx = (vw - sw * scale) / 2;
            ty = (vh - sh * scale) / 2;
            applyTransform();
        }

        // Zoom to cursor (wheel)
        view.addEventListener("wheel", function(e) {
            e.preventDefault();
            var rect = view.getBoundingClientRect();
            var cx = e.clientX - rect.left;
            var cy = e.clientY - rect.top;
            var factor = e.deltaY > 0 ? 0.9 : 1.1;
            var newScale = clampScale(scale * factor);
            tx = cx - (cx - tx) * (newScale / scale);
            ty = cy - (cy - ty) * (newScale / scale);
            scale = newScale;
            applyTransform();
        }, { passive: false });

        // Pan — mouse
        view.addEventListener("mousedown", function(e) {
            dragging = true;
            lastX = e.clientX;
            lastY = e.clientY;
        });
        document.addEventListener("mouseup", function() {
            dragging = false;
        });
        document.addEventListener("mousemove", function(e) {
            if (!dragging) return;
            tx += e.clientX - lastX;
            ty += e.clientY - lastY;
            lastX = e.clientX;
            lastY = e.clientY;
            applyTransform();
        });

        // Pan + pinch — touch
        var activePointers = {};
        var lastPinchDist = 0;

        view.addEventListener("pointerdown", function(e) {
            activePointers[e.pointerId] = { x: e.clientX, y: e.clientY };
            var ids = Object.keys(activePointers);
            if (ids.length === 1) {
                lastX = e.clientX;
                lastY = e.clientY;
            }
            if (ids.length === 2) {
                var a = activePointers[ids[0]], b = activePointers[ids[1]];
                lastPinchDist = Math.hypot(a.x - b.x, a.y - b.y);
            }
        });

        view.addEventListener("pointermove", function(e) {
            if (!activePointers[e.pointerId]) return;
            activePointers[e.pointerId] = { x: e.clientX, y: e.clientY };
            var ids = Object.keys(activePointers);
            if (ids.length === 1) {
                tx += e.clientX - lastX;
                ty += e.clientY - lastY;
                lastX = e.clientX;
                lastY = e.clientY;
                applyTransform();
            } else if (ids.length === 2) {
                var a = activePointers[ids[0]], b = activePointers[ids[1]];
                var dist = Math.hypot(a.x - b.x, a.y - b.y);
                if (lastPinchDist > 0) {
                    var rect = view.getBoundingClientRect();
                    var cx = (a.x + b.x) / 2 - rect.left;
                    var cy = (a.y + b.y) / 2 - rect.top;
                    var newScale = clampScale(scale * (dist / lastPinchDist));
                    tx = cx - (cx - tx) * (newScale / scale);
                    ty = cy - (cy - ty) * (newScale / scale);
                    scale = newScale;
                    applyTransform();
                }
                lastPinchDist = dist;
            }
        });

        view.addEventListener("pointerup", function(e) {
            delete activePointers[e.pointerId];
            if (Object.keys(activePointers).length < 2) lastPinchDist = 0;
        });
        view.addEventListener("pointercancel", function(e) {
            delete activePointers[e.pointerId];
        });

        if (fitBtn) {
            fitBtn.addEventListener("click", function(e) {
                e.stopPropagation();
                fitToView();
            });
        }

        fitToView();
    }

    document.addEventListener("DOMContentLoaded", function() {
        document.querySelectorAll(".canvas-view").forEach(initCanvas);
    });
})();
</script>
```
